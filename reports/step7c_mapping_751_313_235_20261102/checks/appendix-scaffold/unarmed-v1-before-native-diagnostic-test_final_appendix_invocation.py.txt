def put(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    data = value if isinstance(value, bytes) else (json.dumps(value, indent=2) + "\n").encode()
    path.write_bytes(data)
    return data

    raw = put(m.CONFIG, inventory)
    monkeypatch.setattr(m, "CONFIG_SHA256", m.sha(raw))

def test_private_native_apply_preserves_backup_and_completed_journal(case):
    root, evidence, doc = case
    result = m.invoke(root, evidence / "apply.json", apply=True)
    receipt = json.loads((evidence / "apply.json").read_bytes())
    assert result["applied"] is True and receipt["state"] == "complete"
    assert (evidence / "apply.README.before.md").read_bytes() == (
        m.ROOT / m.PARENT / "README.md"
    ).read_bytes()
    assert (root / m.PARENT / "README.md").read_bytes() == Path(
        doc["readme_path"]
    ).read_bytes()
    assert (root / m.PARENT / "final-validation.json").read_bytes() == Path(
        doc["inventory_path"]
    ).read_bytes()
    journal = json.loads(Path(receipt["transaction"]["journal"]).read_bytes())
    assert journal["status"] == "complete"
    assert len(journal["created"]) == 5
    assert journal["changed"] == [str(root / m.PARENT / "README.md")]


@pytest.mark.parametrize("what", ["source", "payload", "readme", "review_pair", "code"])
def test_frozen_input_drift_is_refused_before_publication(case, monkeypatch, what):
    root, evidence, doc = case
    if what == "source":
        path = root / "data/fixture-source.yaml"
    elif what == "payload":
        path = evidence / "full_suite.log"
    elif what == "readme":
        path = root / m.PARENT / "README.md"
    elif what == "review_pair":
        path = root / next(iter(doc["native_review_pair"]))
    else:
        path = evidence / "wrong-code.py"
        put(path, b"old code\n")
        monkeypatch.setattr(m, "CODE", {path: m.sha(b"old code\n")})
    path.write_bytes(path.read_bytes() + b"changed\n")
    with pytest.raises(ValueError, match="Frozen evidence/code changed"):
        m.invoke(root, evidence / "apply.json", apply=True)
    assert not (root / m.PARENT / "checks").exists()
    assert not (evidence / "apply.json").exists()


@pytest.mark.parametrize("change", ["pending", "offscope", "failed", "draft_readme"])
def test_final_configuration_contract_cannot_claim_incomplete_or_out_of_scope_output(
    case, monkeypatch, change
):
    root, evidence, doc = case
    inventory = json.loads(Path(doc["inventory_path"]).read_bytes())
    if change == "pending":
        inventory["pending"].append({"id": "full_suite_v2"})
    elif change == "offscope":
        inventory["files"][0]["proposed_relative_path"] = "../../source.yaml"
    elif change == "failed":
        target = evidence / "full_suite.json"
        receipt = json.loads(target.read_bytes())
        receipt["exit_code"] = 1
        raw = put(target, receipt)
        for item in inventory["files"]:
            if item["source_path"] == str(target):
                item.update(sha256=m.sha(raw), bytes=len(raw))
    else:
        raw = put(Path(doc["readme_path"]), b"FINAL_CHECKS_AND_REVIEW_TO_INSERT\n")
        doc["readme_sha256"] = m.sha(raw)
    raw = put(Path(doc["inventory_path"]), inventory)
    doc["inventory_sha256"] = m.sha(raw)
    raw = put(m.CONFIG, doc)
    monkeypatch.setattr(m, "CONFIG_SHA256", m.sha(raw))
    with pytest.raises(ValueError, match="pending|relative path|command/log|draft"):
        m.invoke(root, evidence / "apply.json", apply=True)
    assert_unpublished(case)


@pytest.mark.parametrize("kind", ["existing", "symlink_parent"])
def test_competing_destinations_are_preserved(case, kind):
    root, evidence, _ = case
    target = root / m.PARENT / "checks"
    if kind == "existing":
        put(target / "fixture/full_suite.log", b"other session evidence\n")
    else:
        foreign = evidence / "foreign"
        foreign.mkdir()
        target.symlink_to(foreign, target_is_directory=True)
    with pytest.raises(ValueError, match="already exists|Aliased"):
        m.invoke(root, evidence / "apply.json", apply=True)
    assert not (evidence / "apply.json").exists()
    if kind == "existing":
        assert (target / "fixture/full_suite.log").read_bytes() == b"other session evidence\n"
    else:
        assert list((evidence / "foreign").iterdir()) == []

@pytest.mark.parametrize("change", ["head", "tracked", "untracked"])
def test_checkout_guard_refuses_unreviewed_checkout_changes(tmp_path, monkeypatch, change):
    def answer(argv, **kwargs):
        if argv[1:3] == ["rev-parse", "HEAD"]:
            return ("other" if change == "head" else m.HEAD).encode()
        if argv[1:3] == ["diff", "--name-only"]:
            return b"data/owner.yaml" if change == "tracked" else b""
        return b"other-session-file\0" if change == "untracked" else b""

    monkeypatch.setattr(m.subprocess, "check_output", answer)
    with pytest.raises(ValueError, match="head changed|preimage changed|native review pair"):
        m.checkout_state(tmp_path, {})
