def require(condition, message):
    if not condition:
        raise ValueError(message)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def read(path):
    require(path.is_absolute() and path == path.resolve(), "Aliased input or destination")
    info = path.lstat()
    require(stat.S_ISREG(info.st_mode) and info.st_nlink == 1, "Nonregular or linked input")
    return path.read_bytes()


def relative(name):
    path = Path(name)
    require(
        not path.is_absolute() and path.parts and ".." not in path.parts and str(path) == name,
        "Unsafe relative path",
    )
    return path


def checkout_state(root, review_pair):
    def git(*args):
        return subprocess.check_output(["git", *args], cwd=root).decode().strip()

    require(git("rev-parse", "HEAD") == HEAD, "Reviewed C5 head changed")
    require(not git("diff", "--name-only", "HEAD", "--"), "Tracked C5 preimage changed")
    untracked = set(
        filter(
            None,
            subprocess.check_output(
                ["git", "ls-files", "--others", "--exclude-standard", "-z"], cwd=root
            )
            .decode()
            .split("\0"),
        )
    )
    require(
        untracked <= set(review_pair),
        "Only the separately saved native review pair and literal diagnostic may be untracked",
    )
    # The diagnostic is intentionally ignored by /reports/*. Check only the
    # exact already hash-bound handoff paths without applying ignore rules;
    # unrelated ignored research/output remains outside this inventory.
    exact_untracked = set()
    if review_pair:
        exact_untracked = set(
            filter(
                None,
                subprocess.check_output(
                    [
                        "git",
                        "--literal-pathspecs",
                        "ls-files",
                        "--others",
                        "-z",
                        "--",
                        *sorted(review_pair),
                    ],
                    cwd=root,
                )
                .decode()
                .split("\0"),
            )
        )
    require(
        exact_untracked == set(review_pair), "Exact native handoff untracked inventory differs"
    )


def invoke(root, log, *, apply=False):
    require(
        CONFIG_SHA256 is not None,
        "Final appendix is unarmed: completed evidence and final freeze are pending",
    )
    require(root.is_absolute() and root == root.resolve(), "Aliased publication root")
    require(
        log.is_absolute() and log.parent == HERE and not os.path.lexists(log),
        "Use a fresh owned external receipt",
    )
    inputs = {}

    def bind(path, expected):
        data = read(path)
        require(sha(data) == expected, "Frozen evidence/code changed: " + str(path))
        inputs[path] = data
        return data

    config = json.loads(bind(CONFIG, CONFIG_SHA256))
    protected = json.loads(bind(PROTECTED, PROTECTED_SHA256))
    require(
        protected["head"] == HEAD and config["head"] == HEAD, "Wrong publication checkpoint"
    )
    require(
        config["scientific_approval_added"] is False
        and config["supervisor_binding"] is None
        and config["documentation_delta_requires_final_head_review"] is True,
        "Invalid authority or final-review qualification",
    )
    for path, digest in CODE.items():
        bind(path, digest)
    invocation_path = Path(__file__).absolute()
    require(read(invocation_path) == _CODE, "Loaded invocation changed")
    inputs[invocation_path] = _CODE
    for name, digest in protected["files"].items():
        bind(root / relative(name), digest)
    inventory_path = Path(config["inventory_path"])
    inventory_raw = bind(inventory_path, config["inventory_sha256"])
    inventory = json.loads(inventory_raw)
    require(
        inventory["state"] == "final_completed_local_validation"
        and inventory["observed_head"] == HEAD
        and inventory["scientific_approval_added"] is False
        and inventory["supervisor_binding"] is None,
        "Inventory is not frozen completed local validation",
    )
    require(
        all(
            item["id"] in {"remote_closure", "final_documentation_delta_review"}
            for item in inventory.get("pending", [])
        ),
        "Required local evidence is still pending",
    )
    require(inventory["destination_parent"] == str(PARENT), "Wrong appendix destination")
    outputs = {}
    by_name = {}
    for item in inventory["files"]:
        name = relative(item["proposed_relative_path"])
        require(
            len(name.parts) >= 2 and name.parts[0] in {"checks", "independent"},
            "Out-of-scope appendix output",
        )
        require(
            name not in outputs and str(name) not in by_name, "Duplicate appendix destination"
        )
        data = bind(Path(item["source_path"]), item["sha256"])
        require(len(data) == item["bytes"], "Frozen evidence size changed")
        outputs[PARENT / name] = data
        by_name[str(name)] = (item, data)
    # Actual completed receipt/log pairs are required. A running log or a status
    # summary cannot stand in for either the final suite or the native pair check.
    require(
        set(config["required_command_pairs"]) == {"full_suite", "native_review_check"},
        "Missing completed local command requirements",
    )
    for pair in config["required_command_pairs"].values():
        receipt_item, receipt_raw = by_name[pair["receipt"]]
        log_item, log_raw = by_name[pair["log"]]
        receipt = json.loads(receipt_raw)
        require(
            receipt["exit_code"] == 0
            and receipt.get("finished_at")
            and receipt["log"] == log_item["source_path"]
            and receipt["log_sha256"] == sha(log_raw),
            "Incomplete or failed command/log binding",
        )
    pair = config["native_review_pair"]
    require(len(pair) == 2, "Exactly the separately saved native review pair is required")
    for name, digest in pair.items():
        bind(root / relative(name), digest)
    require(
        str(DIAGNOSTIC) not in pair, "Diagnostic cannot replace a native review pair member"
    )
    bind(root / DIAGNOSTIC, config["native_review_diagnostic_sha256"])
    native_handoff = set(pair) | {str(DIAGNOSTIC)}
    readme = root / PARENT / "README.md"
    previous = bind(readme, README_BEFORE)
    readme_after = bind(Path(config["readme_path"]), config["readme_sha256"])
    require(
        readme_after != previous and b"FINAL_CHECKS_AND_REVIEW_TO_INSERT" not in readme_after,
        "README is unchanged or still a draft",
    )
    outputs[PARENT / "README.md"] = readme_after
    outputs[PARENT / "final-validation.json"] = inventory_raw
    new_outputs = set(outputs) - {PARENT / "README.md"}

    def scope():
        require(
            (root / PARENT).is_dir() and root / PARENT == (root / PARENT).resolve(),
            "Existing report parent changed",
        )
        require(
            all(root / p == (root / p).resolve() for p in outputs), "Aliased output ancestor"
        )
        require(
            all(not os.path.lexists(root / p) for p in new_outputs),
            "New appendix destination already exists; preserve it",
        )
        require(
            all(read(path) == data for path, data in inputs.items()), "Frozen input changed"
        )
        checkout_state(root, native_handoff)

    scope()
    sys.path.insert(0, str(ROOT))
    from kg_microbe_write import transaction

    from scripts import step7a_guarded_publication as publication

    require(
        Path(transaction.__file__).resolve() in inputs,
        "Loaded native transaction is not the pinned implementation",
    )
    require(
        publication._CODE == inputs[Path(publication.__file__)], "Loaded native helper differs"
    )
    result = {
        "kind": "step7c_final_validation_appendix_publication",
        "stage": "completed_local_validation_before_final_documentation_delta_review",
        "scientific_approval_added": False,
        "supervisor_binding": None,
        "head": HEAD,
        "inventory_sha256": sha(inventory_raw),
        "documentation_delta_requires_final_head_review": True,
        "input_sha256": {str(p): sha(b) for p, b in inputs.items()},
        "output_sha256": {str(p): sha(b) for p, b in outputs.items()},
    }
    backup = log.with_name(log.stem + ".README.before.md")
    answer = publication.publish(
        root,
        outputs,
        inputs,
        log,
        {backup: previous},
        result,
        apply=apply,
        validate_scope=scope,
    )
    if apply:
        require(
            all(read(root / p) == data for p, data in outputs.items()),
            "Published regular output readback differs; retain recovery evidence",
        )
        require(
            all(read(path) == data for path, data in inputs.items() if path != readme),
            "Publication input changed during application; retain recovery evidence",
        )
    return answer
