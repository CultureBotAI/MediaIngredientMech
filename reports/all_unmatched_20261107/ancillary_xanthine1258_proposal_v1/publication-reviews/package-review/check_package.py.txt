def sha(raw): return hashlib.sha256(raw).hexdigest()
def ck(name, ok):
    checks.append({'check': name, 'passed': bool(ok)})
    if not ok: raise AssertionError(name)
def bind(path, expected=None):
    path = Path(path)
    raw = path.read_bytes()
    bindings[str(path)] = sha(raw)
    if expected is not None: ck('exact hash: ' + str(path), sha(raw) == expected)
    return raw
def read(path): return json.loads(bind(path))
def canonical(obj): return json.dumps(obj, sort_keys=True, separators=(',', ':'), ensure_ascii=False, allow_nan=False).encode()
def refs(obj):
    if isinstance(obj, dict):
        if isinstance(obj.get('path'), str) and isinstance(obj.get('sha256'), str): yield obj
        for v in obj.values(): yield from refs(v)
    elif isinstance(obj, list):
        for v in obj: yield from refs(v)
def sensitive_fields(obj, trail=''):
    if isinstance(obj, dict):
        for k, v in obj.items():
            t = trail + '/' + k
            if re.fullmatch(r'(?:access_token|refresh_token|api_key|secret_key|password|authorization|cookie|client_secret|private_key)', k, re.I) and v:
                yield {'field': t, 'value_sha256': sha(canonical(v)), 'value_type': type(v).__name__}
            yield from sensitive_fields(v, t)
    elif isinstance(obj, list):
        for i, v in enumerate(obj): yield from sensitive_fields(v, trail + '/' + str(i))
