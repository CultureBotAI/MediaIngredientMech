import datetime
import json
import os
from pathlib import Path
import subprocess
import sys

name=sys.argv[1];command=sys.argv[2:]
out=D/'checks'/name;out.mkdir(parents=True)
environment={**os.environ,'PYTHONDONTWRITEBYTECODE':'1','GIT_OPTIONAL_LOCKS':'0','PYTHONPATH':str(D/'private/src'),'UV_OFFLINE':'1','UV_NO_SYNC':'1','CLAW_ROOT':str(T/'claw'),'CAMPAIGN_HISTORY_ROOT':str(C/'integration/curation-worktree'),'OPENCLAW_WORKSPACE':str(out/'openclaw-workspace'),'PATH':str(T/'env/bin')+os.pathsep+os.environ.get('PATH','')}

print(json.dumps(receipt,indent=2));sys.exit(result.returncode)
