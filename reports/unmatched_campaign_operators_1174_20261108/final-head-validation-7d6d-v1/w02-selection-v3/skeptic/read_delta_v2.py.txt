def read(p, expected=None):
    p = Path(p)
    assert p.is_absolute() and p.is_file() and not p.is_symlink(), str(p)
    b = p.read_bytes()
    h = hashlib.sha256(b).hexdigest()
    assert expected is None or h == expected, str(p)
    inputs[str(p)] = {'path': str(p), 'sha256': h, 'bytes': len(b)}
    return b


def load(p, expected=None):
    return json.loads(read(p, expected))


def canon(x):
    return json.dumps(x, sort_keys=True, ensure_ascii=False, separators=(',', ':')).encode()

def locrefs(x, pointer=''):
    if isinstance(x, dict):
        if 'absolute_path' in x and 'sha256' in x:
            p = Path(x['absolute_path']); read(p, x['sha256'])
            locator_refs.append({'pointer': pointer, 'path': str(p), 'sha256': x['sha256']})
        for k,v in x.items(): locrefs(v,pointer+'/'+k)
    elif isinstance(x,list):
        for i,v in enumerate(x):locrefs(v,pointer+'/'+str(i))
