def read(path):
    path = Path(path)
    assert stat.S_ISREG(path.lstat().st_mode), path
    assert not any(p.is_symlink() for p in path.parents), path
    return path.read_bytes()


def sha(path):
    return hashlib.sha256(read(path)).hexdigest()


def load(path):
    return json.loads(read(path))


def put(name, value):
    with (D / name).open("x") as stream:
        json.dump(value, stream, indent=2, ensure_ascii=False)
        stream.write("\n")

def add(label, root, path, role, status, expected=None, manifest=None, upstream=None):
    path, root = Path(path), Path(root)
    raw = read(path)
    digest = hashlib.sha256(raw).hexdigest()
    assert expected is None or digest == expected, path
    if upstream:
        assert sha(upstream) == digest, upstream
    rel = path.relative_to(root).as_posix()
    st = path.lstat()
    entry = {
        "packet": label, "packet_root": str(root), "original_path": str(path),
        "packet_relative_path": rel, "sha256": digest, "bytes": len(raw),
        "inode": st.st_ino, "link_count": st.st_nlink,
        "evidence_role": role, "packet_status": status,
        "selection_reason": "Actual sealed operator/fixture/runner evidence; preserves negative history and exact scope; no full-run or application approval",
    }
    if manifest:
        entry["original_manifest_sha256"] = manifest
    if upstream:
        entry["upstream_origin_path"] = str(upstream)
        entry["upstream_exact_bytes_rechecked"] = True
    if digest in known:
        entry.update(known[digest])
    else:
        target = f"{ARCH}/{label}/{rel}" + (".txt" if path.suffix == ".py" else "")
        entry.update(disposition="copy_exact_bytes_when_authorized", archive_path=target)
        known[digest] = {"disposition": "reuse_selected_new_archive_bytes", "archive_path": target}
    entries.append(entry)


def package(label, root, manifest_sha, role, status, outer=None, outer_sha=None, size_key="bytes"):
    root = Path(root)
    manifest = root / "MANIFEST.json"
    assert sha(manifest) == manifest_sha
    doc = load(manifest)
    members = doc["files"]
    assert len({x["path"] for x in members}) == len(members)
    for item in members:
        path = root / item["path"]
        assert size_key in {"bytes", "size"}
        assert type(item[size_key]) is int and len(read(path)) == item[size_key]
        add(label, root, path, role, status, item["sha256"], manifest_sha, item.get("original_path"))
    add(label, root, manifest, role + "_original_complete_manifest", status, manifest_sha)
    if outer:
        add(label + "-outer", Path(outer).parent, outer, role + "_original_scope_handoff", status, outer_sha)
    verified_manifests.append({"path": str(manifest), "sha256": manifest_sha, "members": len(members)})
    packets.append({"packet": label, "root": str(root), "role": role, "status": status, "manifest": str(manifest), "manifest_sha256": manifest_sha, "members": len(members), "complete_declared_portable_members_selected_or_exactly_deduplicated": True, "full_runtime_fixture_claim": False})
    omissions.append({"packet": label, "policy": "Every declared portable member and complete original manifest is retained or hash-deduplicated. Runtime Git repos, full corpora, caches, environments, credentials and TUI/session data are excluded. Original complete manifests describe original packets, not a complete relocated execution tree.", "original_omission_policy": doc.get("omission_policy", doc.get("scope"))})
