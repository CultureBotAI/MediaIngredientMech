"""Create-only closeout of the reviewed private packet; no archive operation."""
from datetime import datetime, timezone
from pathlib import Path
import hashlib
import json
import os
import re
import subprocess
import sys

O = Path(__file__).resolve().parent
W = O.parent.parent
K = W.parents[2] / "integration/release-worktree"
OUTER = {"MANIFEST.json", "SEAL.json", "SHA256SUMS"}


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def load(name):
    return json.loads((O / name).read_text())


def check(base, row):
    path = base / row["path"]
    if row.get("kind") == "diagnostic_symlink" or "target" in row:
        assert path.is_symlink() and os.readlink(path) == row["target"], path
    else:
        assert path.is_file() and not path.is_symlink(), path
        assert sha(path) == row["sha256"], path
        if "bytes" in row:
            assert path.stat().st_size == row["bytes"], path


def inventory():
    rows, directories = [], []
    for path in sorted(O.rglob("*")):
        name = str(path.relative_to(O))
        if name in OUTER:
            continue
        if path.is_symlink():
            rows.append({"path": name, "kind": "diagnostic_symlink", "target": os.readlink(path)})
        elif path.is_file():
            rows.append({"path": name, "kind": "file", "sha256": sha(path), "bytes": path.stat().st_size})
        elif path.is_dir():
            directories.append(name)
        else:
            raise AssertionError(f"Unexpected private member: {path}")
    return rows, directories


def write(name, obj):
    with (O / name).open("x") as stream:
        stream.write(json.dumps(obj, indent=2) + "\n")
