"""Read-only final Git/blob bridge; writes only new evidence in this reviewer lane."""
from pathlib import Path
import ast
import datetime
import difflib
import hashlib
import json
import os
import subprocess

O = Path(__file__).resolve().parent
W = O.parents[3]
C = W.parents[2]
K = C / "integration/release-worktree"
HEAD = "7d6d44ea132e04709f81ebd9e1577643745110ad"
PARENT = "e0a75a0c48ba539426b1f70623ea376c7b379ecf"
environment = dict(os.environ, GIT_OPTIONAL_LOCKS="0", PYTHONDONTWRITEBYTECODE="1")
commands = []


def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def sha(data):
    return hashlib.sha256(data).hexdigest()


def ref(path):
    data = path.read_bytes()
    return {"path": str(path), "sha256": sha(data), "bytes": len(data)}


def git(*args):
    argv = ["git", "-C", str(K), *args]
    p = subprocess.run(argv, capture_output=True, env=environment, check=True)
    commands.append({"at": now(), "argv": argv, "exit_code": p.returncode,
                     "stdout_sha256": sha(p.stdout), "stdout_bytes": len(p.stdout),
                     "stderr": p.stderr.decode()})
    return p.stdout


def dump_new(name, obj):
    with (O / name).open("x") as handle:
        json.dump(obj, handle, indent=2)
        handle.write("\n")


started = now()
prep = json.loads((O / "PREPARATORY-MANIFEST.json").read_text())
for entry in prep["files"]:
    p = O / entry["path"]
    assert sha(p.read_bytes()) == entry["sha256"], entry
    assert p.stat().st_size == entry["bytes"], entry

commit_receipt_path = C / "preflight/campaign-suppressions1259-commit-v1/receipt.json"
commit_receipt = json.loads(commit_receipt_path.read_text())
assert commit_receipt["head"] == HEAD and commit_receipt["parent"] == PARENT
assert git("rev-parse", "HEAD").decode().strip() == HEAD
assert git("show", "-s", "--format=%P", HEAD).decode().strip() == PARENT
assert git("status", "--porcelain").decode() == ""
changed = git("diff-tree", "--no-commit-id", "--name-only", "-r", HEAD).decode().splitlines()
assert set(changed) == set(commit_receipt["files"])

pure = json.loads((O / "pure-controls-result.json").read_text())
paths = {item["path"] for item in pure["executed_producers"]}
paths |= set(commit_receipt["files"])
paths |= {"scripts/publish_unmatched_campaign.py", "scripts/validate_unmatched_campaign_publication.py",
          "tests/test_campaign_owner_apply.py", "tests/test_campaign_demotion_integration.py",
          "MAPPING_SEMANTICS.md", "AGENTS.md"}
blobs = {}
for path in sorted(paths):
    data = git("show", f"{HEAD}:{path}")
    current = (K / path).read_bytes()
    assert current == data, path
    blobs[path] = {"sha256": sha(data), "bytes": len(data), "current_K_equals_final_git_blob": True}
for item in pure["executed_producers"]:
    assert blobs[item["path"]]["sha256"] == item["sha256"], item
for path, expected in commit_receipt["files"].items():
    assert blobs[path]["sha256"] == expected, path
assert blobs["src/mediaingredientmech/campaign_publication.py"]["sha256"] == sha((O / "inputs/campaign_publication-candidate.py").read_bytes())
copy_pairs = {
    "scripts/publish_unmatched_campaign.py": "publish_unmatched_campaign.py",

test_diffs = {}
for path in ["tests/test_campaign_owner_apply.py", "tests/test_campaign_publication.py", "tests/test_campaign_demotion_integration.py"]:
    before = git("show", f"{PARENT}:{path}").decode()
    after = (K / path).read_text()
    b, a = ast.parse(before), ast.parse(after)
    asserts = lambda t: [ast.dump(n, include_attributes=False) for n in ast.walk(t) if isinstance(n, ast.Assert)]
    decorators = lambda t: {n.name: [ast.dump(d, include_attributes=False) for d in n.decorator_list] for n in t.body if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef))}
    assert asserts(a) == asserts(b), path
    assert decorators(a) == decorators(b), path
    diff = ''.join(difflib.unified_diff(before.splitlines(True), after.splitlines(True), fromfile=PARENT + ':' + path, tofile=HEAD + ':' + path))
    test_diffs[path] = {"before_sha256": sha(before.encode()), "after_sha256": sha(after.encode()),
                        "existing_asserts_exact": True, "function_decorators_exact": True, "diff": diff}
    if path.endswith("test_campaign_publication.py"):
        reconstructed = after.replace('        "scripts/record_review.py",\n        "schema/record_review.yaml",\n', '', 1)
        assert reconstructed == before
        test_diffs[path]["only_change_exact_two_real_dependency_paths"] = True
    else:
        assert before == after

runtime_sources = [
