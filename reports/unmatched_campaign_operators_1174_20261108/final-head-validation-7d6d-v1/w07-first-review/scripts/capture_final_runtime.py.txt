from pathlib import Path
import datetime, hashlib, json, shutil

O = Path(__file__).resolve().parents[1]
W = O.parents[1]
C = W.parents[2]
S = C / 'preflight/final-suppression-7d6d44ea-validation-v1'
D = O / 'inputs/final-runtime-observation-v1'
D.mkdir()
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
now = lambda: datetime.datetime.now(datetime.timezone.utc).isoformat()
files = []
for name in ['runtime.json', 'qc-command.json', 'qc-process.json', 'qc-receipt.json', 'qc.log', 'test-command.json', 'test-process.json']:
    p = S / name
    b = p.read_bytes()
    q = D / name
    q.write_bytes(b)
    assert q.read_bytes() == p.read_bytes()
    files.append({'original_path': str(p), 'copy': str(q.relative_to(O)), 'sha256': sha(q), 'accessed_at': now(), 'bytes': len(b)})
before = json.loads((S / 'before.json').read_text())
after = json.loads((S / 'qc-after.json').read_text())
receipt = json.loads((D / 'qc-receipt.json').read_text())
assert receipt['exit_code'] == 0
assert receipt['head'] == before['head'] == after['head'] == '7d6d44ea132e04709f81ebd9e1577643745110ad'
assert receipt['log_sha256'] == sha(D / 'qc.log')
comparison = {k: before[k] == after[k] for k in before if k != 'at'}
assert all(comparison.values())
assert comparison == receipt['input_equivalence']
references = [{'path': str(S / n), 'sha256': sha(S / n), 'bytes': (S / n).stat().st_size} for n in ['before.json', 'qc-after.json']]
members = sorted(p.name for p in S.iterdir())
suite = 'PENDING_NO_COMPLETED_TEST_RECEIPT_OBSERVED' if 'test-receipt.json' not in members else 'RECEIPT_EXISTS_REQUIRES_SEPARATE_REVIEW'
out = {'at': now(), 'scope': 'Read-only observation of actual root-run receipts; no W07 execution of qc/test and no application approval.', 'head': receipt['head'], 'actual_runtime_cwd': json.loads((D/'qc-command.json').read_text())['cwd'], 'qc': {'exit_code': 0, 'log_sha256': sha(D/'qc.log'), 'input_equivalence_independently_recomputed': comparison}, 'copied_files': files, 'referenced_complete_input_inventories': references, 'suite_status': suite, 'directory_members_including_ignored': members, 'runtime_limit': 'QC is the unchanged pre-application corpus at this code head. It does not attest a later actual 934-operation application or publication.'}
(O/'validation/final-runtime-observation-v1.json').write_text(json.dumps(out,indent=2)+'\n')
print(json.dumps({'qc_exit': 0, 'suite': suite, 'actual_cwd': out['actual_runtime_cwd']}))
