def sha(p):
    h=hashlib.sha256()
    with p.open('rb') as f:
        for b in iter(lambda:f.read(1024*1024),b''):h.update(b)
    return h.hexdigest()
def ref(p):return {'path':str(p),'sha256':sha(p),'size':p.stat().st_size}
def js(p):return json.loads(p.read_text())
def git(p,*args):
    q=subprocess.run(['git','-C',str(p),*args],capture_output=True,text=True,env=os.environ|{'GIT_OPTIONAL_LOCKS':'0'})
    return {'command':['git','-C',str(p),*args],'exit':q.returncode,'stdout':q.stdout,'stderr':q.stderr}
