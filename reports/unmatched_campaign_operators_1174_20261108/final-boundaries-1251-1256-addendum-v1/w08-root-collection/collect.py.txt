def digest(p):
    before = p.lstat()
    assert stat.S_ISREG(before.st_mode), (p, "not regular")
    fd = os.open(p, os.O_RDONLY | os.O_NOFOLLOW)
    h = hashlib.sha256()
    with os.fdopen(fd, "rb") as stream:
        opened = os.fstat(stream.fileno())
        assert (opened.st_dev, opened.st_ino) == (before.st_dev, before.st_ino)
        while chunk := stream.read(1024 * 1024):
            h.update(chunk)
        end = os.fstat(stream.fileno())
    after = p.lstat()
    signature = lambda s: (s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns)
    assert signature(before) == signature(end) == signature(after), (p, "changed")
    return h.hexdigest()

def member_path(root, spelling):
    rel = PurePosixPath(spelling)
    assert spelling == rel.as_posix() and not rel.is_absolute() and ".." not in rel.parts
    current = root
    for part in rel.parts[:-1]:
        current /= part
        assert stat.S_ISDIR(current.lstat().st_mode), (current, "aliased parent")
    return root / spelling

def verify_manifest(path, complete=False):
    obj, root = read(path), path.parent
    entries = obj.get("members", obj.get("files"))
    if isinstance(entries, dict):
        entries = [{"path": k, "sha256": v} for k, v in entries.items()]
    assert isinstance(entries, list), path
    files = [x for x in entries if x.get("kind", "file") == "file"]
    links = [x for x in entries if x.get("kind") == "diagnostic_symlink"]
    links += obj.get("symlinks", obj.get("literal_symlinks_not_followed", []))
    assert len({x["path"] for x in files + links}) == len(files + links)
    def regular(entry):
        p = member_path(root, entry["path"])
        actual = digest(p)
        assert actual == entry["sha256"], (p, actual, entry["sha256"])
        expected_size = entry.get("bytes", entry.get("size"))
        if expected_size is not None:
            assert p.stat().st_size == expected_size, p
        return {"path": entry["path"], "sha256": actual}
    with ThreadPoolExecutor(max_workers=4) as pool:
        verified = list(pool.map(regular, files))
    for entry in links:
        p = member_path(root, entry["path"])
        assert stat.S_ISLNK(p.lstat().st_mode), p
        assert os.readlink(p) == entry.get("target", entry.get("literal_target")), p
    hardlinks = obj.get("hardlink_groups", obj.get("private_fixture_hardlinks", []))
    for entry in hardlinks:
        paths = entry.get("paths") or ([m["path"] for m in entry["members"]] if "members" in entry else [entry["path"]])
        states = [member_path(root, p).lstat() for p in paths]
        expected_links = entry.get("link_count", entry.get("members", [{}])[0].get("nlink"))
        assert all(s.st_nlink == expected_links for s in states)
        assert len({(s.st_dev, s.st_ino) for s in states}) == 1
        if "device" in entry:
            assert all((s.st_dev, s.st_ino) == (entry["device"], entry["inode"]) for s in states)
    extra = []
    if complete:
        actual = set()
        for base, dirs, names in os.walk(root, followlinks=False):
            for name in dirs[:]:
                p = Path(base) / name
                if p.is_symlink():
                    actual.add(str(p.relative_to(root)))
                    dirs.remove(name)
            actual.update(str((Path(base) / name).relative_to(root)) for name in names)
        declared = {x["path"] for x in files + links}
        extra = sorted(actual - declared)
        assert set(extra) <= {"MANIFEST.json", "MANIFEST.sha256", "SEAL.json"}, extra
    return {"manifest": str(path), "sha256": digest(path), "regular_files": len(verified),
            "literal_symlinks": len(links), "hardlink_metadata_entries": len(hardlinks),
            "all_members_verified": True, "ignored_and_hidden_included": True,
            "symlinks_followed": False, "extra_root_seal_files": extra}
