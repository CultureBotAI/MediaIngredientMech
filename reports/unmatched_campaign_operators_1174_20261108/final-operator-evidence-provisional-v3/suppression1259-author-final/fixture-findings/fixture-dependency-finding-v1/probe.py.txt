import ast
import hashlib
import json
import os
import shutil
import subprocess
from datetime import datetime, timezone
from pathlib import Path

D = Path(__file__).absolute().parent
P = D / "runtime/repo"
R = D.parents[3]
PY = R / ".temp/kgm-unmapped-20261107/env/bin/python"
OUT = D / "fixture-dependency-finding-v1"
OUT.mkdir(exist_ok=False)
sha = lambda raw: hashlib.sha256(raw).hexdigest()
test = P / "tests/test_campaign_publication.py"
publisher = P / "src/mediaingredientmech/campaign_publication.py"
before = D / "before/src/mediaingredientmech/campaign_publication.py"
failed = D / "checks/preload-single-diagnostic-v1/fixtures/test_1195_preloaded_public_dep0/private-package"

env = dict(os.environ, PYTHONDONTWRITEBYTECODE="1", PYTEST_ADDOPTS="", UV_OFFLINE="1")
results = []

def run(name, argv, root):
    local_env = dict(env, PYTHONPATH=str(root / "src"))
    started = datetime.now(timezone.utc).isoformat()
    result = subprocess.run(argv, cwd=root, env=local_env, capture_output=True)
    (OUT / (name + ".stdout")).write_bytes(result.stdout)
    (OUT / (name + ".stderr")).write_bytes(result.stderr)
    receipt = {"name": name, "argv": argv, "cwd": str(root), "started_at": started,
               "ended_at": datetime.now(timezone.utc).isoformat(), "exit_code": result.returncode,
               "PYTHONPATH": local_env["PYTHONPATH"], "stdout_sha256": sha(result.stdout),
               "stderr_sha256": sha(result.stderr)}
    results.append(receipt)
    print(json.dumps(receipt), flush=True)
    return result

actual = run("frozen-import-full-trace", [str(PY), "-B", "-c", "from mediaingredientmech import campaign_publication"], failed)
assert actual.returncode != 0 and b"scripts/record_review.py" in actual.stderr
