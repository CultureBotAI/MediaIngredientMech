def now():
    return datetime.datetime.now(datetime.UTC).isoformat()


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def require(condition, message):
    if not condition:
        raise RuntimeError(message)


def save(path, value):
    with path.open("x") as handle:
        json.dump(value, handle, indent=2)
        handle.write("\n")


def git(root, *args):
    return subprocess.check_output(
        ["git", "-C", str(root), *args],
        env={**os.environ, "GIT_OPTIONAL_LOCKS": "0"},
    )


def head(root):
    return git(root, "rev-parse", "HEAD").decode().strip()


def clean(root):
    return git(root, "status", "--porcelain=v1", "--untracked-files=all").decode()

def tracked_snapshot(root):
    result = {}
    for raw in git(root, "ls-files", "-z").split(b"\0"):
        if not raw:
            continue
        name = os.fsdecode(raw)
        path = root / name
        if not os.path.lexists(path):
            result[name] = {"missing": True}
            continue
        info = path.lstat()
        if stat.S_ISLNK(info.st_mode):
            result[name] = {"symlink": os.readlink(path)}
        else:
            require(stat.S_ISREG(info.st_mode), "Unexpected tracked file type: " + name)
            result[name] = {"sha256": sha(path), "executable": bool(info.st_mode & 0o111)}
    return result


def snapshot(output):
    # Direct-owner enumeration includes gitignored files, not nested backups.
    owners = {}
    for bucket in ("mapped", "unmapped"):
        for path in sorted((I / "data/ingredients" / bucket).glob("*.yaml")):
            require(not path.is_symlink(), "Owner alias: " + str(path))
            owners[path.relative_to(I).as_posix()] = sha(path)
    producer_paths = {
        T / "claw/scripts/build_mim_ingredient_sssom.py",
        T / "claw/scripts/kgm_unified_mappings.py",
        T / "claw/plugins/lock_manager.py",
    }
    for package in ("kg_microbe_utils", "kg_microbe_write", "kg_microbe_fleet"):
        producer_paths.update((T / "claw/src" / package).rglob("*.py"))
    env = {**os.environ, **environment(output)}
    command = [str(I / ".venv/bin/python"), "-B", "-c", (
        "import json;from pathlib import Path;"
        "from mediaingredientmech.validation.mapping_change_receipts import verify_mapping_chain;"
        f"p=verify_mapping_chain(Path({str(J)!r}));"
        "assert len(p['receipts'])==27;"
        "print(json.dumps({'inputs':p['input_hashes'],'receipt_inputs':p['receipt_inputs']}))"
    )]
    history = json.loads(subprocess.check_output(command, cwd=I, env=env))
    return {
        "at": now(), "head": head(I), "status": clean(I),
        "tracked": tracked_snapshot(I), "direct_owner_sha256": owners,
        "claw_head": head(T / "claw"), "claw_status": clean(T / "claw"),
        "producer_sha256": {str(p): sha(p) for p in sorted(producer_paths)},
        "history_head": head(J), "history_status": clean(J), "history": history,
    }


def verify_start(expected_head):
    require(re.fullmatch(r"[0-9a-f]{40}", expected_head), "Explicit full commit SHA required")
    require(expected_head not in OLD_HEADS, "Do not repeat a superseded validation checkpoint")
    require(head(I) == expected_head, "I has not been fast-forwarded to the expected head")
    require(not clean(I), "I must be clean before native validation")
    expected = json.loads((HERE / "expected-inputs.json").read_bytes())
    for name, digest in expected["expected_final_code"].items():
        require(sha(I / name) == digest, "Final corrected code differs: " + name)
    require(head(J) == J_HEAD and not clean(J), "Frozen historical checkout changed")
    require(head(T / "claw") == CLAW_HEAD, "Pinned native producer revision changed")
    require(not clean(T / "claw"), "Native producer checkout must be clean")


def equivalent(before, after):
    return {key: before[key] == after[key] for key in before if key != "at"}


def integration_marks(log, filename):
    """Keep the original full-dot requirement for every pinned native integration file."""
    lines = log.splitlines()
    starts = [n for n, line in enumerate(lines) if line.startswith(filename + " ")]
    require(len(starts) == 1, "Missing or repeated native integration file: " + filename)
    first = starts[0]
    block = [lines[first].split(".py ", 1)[1]]
    for line in lines[first + 1:]:
        if line.startswith("tests/"):
            break
        block.append(line)
    return "".join(re.sub(r"\[.*?\]", "", line).strip() for line in block)


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("mode", choices=("capture", "qc", "test", "seal"))
    parser.add_argument("--head", required=True)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--readiness-evidence", type=Path)
    parser.add_argument("--readiness-sha256")
    args = parser.parse_args(argv)
    verify_start(args.head)
    output = args.output.absolute()
    require(".." not in output.parts, "Noncanonical output path refused")
    require(output.is_relative_to(C / "preflight"), "Output must stay under task preflight")
    require(not any(p.is_symlink() for p in (output, *output.parents)), "Aliased output refused")
    if args.mode == "capture":
        require(args.readiness_evidence is not None and args.readiness_sha256 is not None,
                "Pin root-confirmed independent readiness evidence before capture")
        evidence = args.readiness_evidence.absolute()
        require(sha(evidence) == args.readiness_sha256, "Readiness evidence bytes differ")
        output.mkdir(exist_ok=False)
        (output / "temporary").mkdir()
        (output / "private-openclaw-workspace").mkdir()
        before = snapshot(output)
        save(output / "before.json", before)
        env = environment(output)
        run = subprocess.run(["/Users/marcin/.local/bin/uv", "run", "--frozen", "python", "-c",
                              "import sys;assert sys.version_info[:2]==(3,13);print(sys.executable);print(sys.version)"],
                             cwd=I, env={**os.environ, **env}, capture_output=True, text=True)
        save(output / "runtime.json", {"exit": run.returncode, "stdout": run.stdout, "stderr": run.stderr,
             "environment": env, "readiness_evidence": str(evidence), "readiness_sha256": sha(evidence),
             "readiness_interpretation": "Root confirms initial independent residual probes; this runner only pins that evidence, not a new approval gate",
             "runner_sha256": sha(Path(__file__)), "expected_inputs_sha256": sha(HERE / "expected-inputs.json")})
        require(run.returncode == 0, "Maintained runtime check failed")
        return
    before = json.loads((output / "before.json").read_bytes())
    require(before["head"] == args.head, "Captured baseline differs")
    runtime = json.loads((output / "runtime.json").read_bytes())
    require(runtime["runner_sha256"] == sha(Path(__file__)), "Captured runner changed")
    require(runtime["expected_inputs_sha256"] == sha(HERE / "expected-inputs.json"), "Expected code pins changed")
    if args.mode in ("qc", "test"):
        env = environment(output)
        start = snapshot(output)
        require(all(equivalent(before, start).values()), "Inputs changed before validation")
        command = ["/opt/homebrew/bin/just", args.mode]
        save(output / (args.mode + "-command.json"), {"at": now(), "argv": command, "cwd": str(I), "environment": env, "head": args.head})
        with (output / (args.mode + ".log")).open("xb") as log:
            process = subprocess.Popen(command, cwd=I, env={**os.environ, **env}, stdout=log, stderr=subprocess.STDOUT)
            save(output / (args.mode + "-process.json"), {"pid": process.pid, "parent_pid": os.getpid(), "argv": command, "head": args.head})
            exit_code = process.wait()
        after = snapshot(output)
        save(output / (args.mode + "-after.json"), after)
        receipt = {"at": now(), "head": after["head"], "exit_code": exit_code,
                   "input_equivalence": equivalent(before, after), "log_sha256": sha(output / (args.mode + ".log"))}
        save(output / (args.mode + "-receipt.json"), receipt)
        require(all(receipt["input_equivalence"].values()), "Input drift preserved; stop without repair")
        raise SystemExit(exit_code)
    receipts = {mode: json.loads((output / (mode + "-receipt.json")).read_bytes()) for mode in ("qc", "test")}
    require(all(r["exit_code"] == 0 and all(r["input_equivalence"].values()) for r in receipts.values()),
            "Cannot seal a failed/interrupted full validation as PASS")
    final = snapshot(output)
    require(all(equivalent(before, final).values()), "Final input drift")
    log = (output / "test.log").read_text()
    result = [line for line in log.splitlines() if re.search(r"\d+ passed.* in \d+", line)]
    require(len(result) == 1, "Missing unambiguous complete pytest summary")
    expected_counts = json.loads((HERE / "expected-inputs.json").read_bytes())["required_native_cases"]
    native_counts = {}
    for filename, expected_count in expected_counts.items():
        require(type(expected_count) is int and expected_count > 0, "Invalid native case count")
        marks = integration_marks(log, filename)
        require(marks == "." * expected_count,
                f"Every required native integration case must execute and pass: {filename}")
        native_counts[filename] = {"passed": expected_count, "skipped": 0}
    summary = {"head": args.head, "scope": "Author native full validation; not scientific/application/merge approval",
               "pytest": result[0], "real_generator_cases_passed": sum(v["passed"] for v in native_counts.values()),
               "real_generator_cases_skipped": 0, "required_native_files": native_counts,
               "coverage_lines": [line for line in lines if line.startswith("TOTAL") or "fail-under" in line],
               "B4_lines": [line for line in (output / "qc.log").read_text().splitlines() if "B4" in line],
               "all_tracked_inputs": len(before["tracked"]), "direct_owner_count": len(before["direct_owner_sha256"]),
               "all_input_membership_and_bytes_unchanged": True, "receipts": receipts}
    save(output / "SUMMARY.json", summary)
    manifest = {p.name: sha(p) for p in sorted(output.iterdir()) if p.is_file()}
    save(output / "MANIFEST.json", {"files": manifest, "private_temporary_trees_excluded": True})
    save(output / "HANDOFF.json", {"head": args.head, "status": "AUTHOR_FULL_VALIDATION_PASS",
         "manifest_sha256": sha(output / "MANIFEST.json"), "summary_sha256": sha(output / "SUMMARY.json"),
         "scientific_or_merge_approval": False})


if __name__ == "__main__":
    main()
