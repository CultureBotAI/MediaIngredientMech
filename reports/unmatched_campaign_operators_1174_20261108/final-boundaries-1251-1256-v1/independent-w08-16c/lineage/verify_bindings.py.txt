def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def js(p):
    return json.loads(p.read_bytes())

def ref(p):
    return {'path': str(p), 'sha256': sha(p.read_bytes()), 'size': p.stat().st_size}

checks = []

def check(name, passed, **details):
    checks.append({'check': name, 'passed': bool(passed), **details})

def command(root, *args):
    result = subprocess.run(['git', '-C', str(root), *args], capture_output=True,
                            env=os.environ | {'GIT_OPTIONAL_LOCKS': '0'})
    return {'command': ['git', '-C', str(root), *args], 'exit_code': result.returncode,
            'stdout': result.stdout.decode(), 'stderr': result.stderr.decode()}

def funcs(p):
    return {n.name:ast.dump(n,include_attributes=False) for n in ast.parse(p.read_text()).body
            if isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef))}

 ('publisher_prior',B/'before/tests/test_campaign_publication.py',F/'tests/test_campaign_publication.py')]:
    oldfunctions=funcs(old); newfunctions=funcs(new)
    exact={k:newfunctions.get(k)==v for k,v in oldfunctions.items() if k.startswith('test_')}
    ast_checks.append({'label':label,'before':ref(old),'after':ref(new),'test_functions':exact,
                       'complete_old_file_byte_prefix':new.read_bytes().startswith(old.read_bytes())})
