def read(path):
    return json.loads(path.read_text())

def sha(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def ref(name):
    path = P / name
    assert path.is_file() and not path.is_symlink()
    return {'path': str(path), 'sha256': sha(path)}

def save(path, document):
    with path.open('x') as stream:
        json.dump(document, stream, indent=2, ensure_ascii=False)
        stream.write('\n')

members, links, hardlinks = [], [], []
for parent, dirs, files in os.walk(P, followlinks=False):
    for name in dirs + files:
        path = Path(parent) / name
        info = path.lstat()
        relative = str(path.relative_to(P))
        if stat.S_ISLNK(info.st_mode):
            links.append({'path': relative, 'literal_target': os.readlink(path)})
        elif stat.S_ISREG(info.st_mode) and relative not in {'MANIFEST.json', 'SEAL.json'}:
            members.append({'path': relative, 'sha256': sha(path), 'size': info.st_size})
            if info.st_nlink > 1:
                hardlinks.append({'path': relative, 'device': info.st_dev, 'inode': info.st_ino,
                                  'link_count': info.st_nlink})
manifest = {'schema_version': 1, 'members': sorted(members, key=lambda x: x['path']),
