def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def save(name, value):
    with (O / name).open('x') as f:
        json.dump(value, f, indent=2)
        f.write('\n')

assert Path(owner.__file__).resolve() == F / 'src/mediaingredientmech/curate/campaign_owner_apply.py'
assert Path(publication.__file__).resolve() == F / 'src/mediaingredientmech/campaign_publication.py'
assert subprocess.check_output(['git', '-C', str(CLAW), 'rev-parse', 'HEAD'], text=True).strip() == publication.PRODUCER_REVISION

class Recorder:
    def __init__(self):
        self.collected = []
        self.reports = []

    def pytest_collection_finish(self, session):
        self.collected = [item.nodeid for item in session.items]

    def pytest_runtest_logreport(self, report):
        self.reports.append({'nodeid': report.nodeid, 'when': report.when,
                             'outcome': report.outcome, 'duration': report.duration,
                             'longrepr': str(report.longrepr) if report.longrepr else None})

recorder = Recorder()
code = int(pytest.main(args, plugins=[recorder]))
loaded = {}
for name, module in sorted(sys.modules.items()):
    f = getattr(module, '__file__', None)
    if f and (name.startswith('mediaingredientmech') or name.startswith('test_campaign_')):
        path = Path(f).resolve()
        if path.is_file():
            loaded[name] = {'path': str(path), 'sha256': sha(path)}
integration = [r for r in recorder.reports if 'test_campaign_demotion_integration.py' in r['nodeid'] and r['when'] == 'call']
summary = {'exit_code': code, 'started_at': started,
           'finished_at': datetime.datetime.now(datetime.timezone.utc).isoformat(),
           'collected': len(recorder.collected),
           'passed_calls': sum(r['when'] == 'call' and r['outcome'] == 'passed' for r in recorder.reports),
           'skips': [r for r in recorder.reports if r['outcome'] == 'skipped'],
           'failures': [r for r in recorder.reports if r['outcome'] == 'failed'],
           'integration_executed': len(integration),
           'integration_passed': sum(r['outcome'] == 'passed' for r in integration),
           'actual_loaded_files': loaded}
save('collected.json', recorder.collected)
save('test-reports.json', recorder.reports)
save('result.json', summary)
print(json.dumps({k: v for k, v in summary.items() if k != 'actual_loaded_files'}, indent=2))
assert code == 0 and summary['collected'] == 337 and summary['passed_calls'] == 337
assert summary['integration_executed'] == summary['integration_passed'] == 9
assert not summary['skips'] and not summary['failures']
