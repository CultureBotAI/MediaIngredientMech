def check(name, fn, error=None):
    try:
        value=fn()
    except (ValueError,RuntimeError,FileNotFoundError) as exc:
        result={'name':name,'accepted':False,'error_type':type(exc).__name__,'message':str(exc),'expected_rejection':error is not None,'pass':error is not None and error in str(exc)}
    else:
        result={'name':name,'accepted':True,'result':value,'expected_rejection':error is not None,'pass':error is None}
    results.append(result)
    (P/'probe-progress.json').write_text(json.dumps(results,indent=2)+'\n')
    assert result['pass'],result
