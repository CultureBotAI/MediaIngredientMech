from pathlib import Path
import datetime
import hashlib
import json
import tarfile
import xml.etree.ElementTree as ET

D = Path(__file__).absolute().parent
C = D.parents[1]
P = D.parent / "unicode-application1255-fix-v1"
def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
def read(path):
    return json.loads(path.read_text())
def save(name, value):
    with (D / name).open("x") as stream:
        json.dump(value, stream, indent=2)
        stream.write("\n")
def counts(name):
    totals = {key: 0 for key in ("tests", "errors", "failures", "skipped")}
    for suite in ET.parse(D / name).getroot().iter("testsuite"):
        for key in totals:
            totals[key] += int(suite.get(key, "0"))
    totals["passed"] = totals["tests"] - sum(totals[key] for key in ("errors", "failures", "skipped"))
    return totals

focused = counts("pytest-head-v2.xml")
combined = counts("pytest-combined-v1.xml")

assert combined == {"tests": 441, "errors": 0, "failures": 0, "skipped": 0, "passed": 441}, combined

for path in sorted(D.rglob("*")):
    if not path.is_file():
        continue
    rel = path.relative_to(D)
    if any(part.startswith("pytest-") for part in rel.parts[:-1]) or "mypy-cache-v1" in rel.parts or ".git" in rel.parts:
