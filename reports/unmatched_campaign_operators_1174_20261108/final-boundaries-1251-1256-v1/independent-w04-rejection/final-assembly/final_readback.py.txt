def xmlrows(path):
    root=ET.fromstring(read(path));out=[]
    for node in root.iter('testcase'):
        status='error' if node.find('error') is not None else 'failed' if node.find('failure') is not None else 'skipped' if node.find('skipped') is not None else 'passed'
        row={'classname':node.attrib['classname'],'name':node.attrib['name'],'time_seconds':float(node.attrib['time']),'status':status}
        issue=node.find('error')
        if issue is None:issue=node.find('failure')
        if issue is not None:
            trace=issue.text or ''
            row.update(message=issue.attrib.get('message'),error_type=issue.attrib.get('type'),traceback_sha256=sha(trace.encode()),missing_historical_maker='make_receipt.py' in trace and 'FileNotFoundError' in trace)
        out.append(row)
    return out
