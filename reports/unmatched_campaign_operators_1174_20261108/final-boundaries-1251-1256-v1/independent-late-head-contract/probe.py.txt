def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

def commit(root, label, unrelated=False, include_owners=False):
    if unrelated:
        (root / "unrelated-after-acceptance.txt").write_text("Unrelated committed fixture data.\n")
        fixture.run_git(root, "add", "unrelated-after-acceptance.txt")
    if include_owners:
        fixture.run_git(root, "add", "data/ingredients")
    fixture.run_git(root, "-c", "user.email=fixture@example.invalid", "-c",
                    "user.name=Independent transaction review", "commit", "--allow-empty", "-qm", label)

             "historical_later_empty", "historical_committed_owners_and_unrelated",
             "historical_tampered_current_owner"]:
    root = OUT / "fixtures" / ("source-" + mode)
    root.mkdir(parents=True, exist_ok=False)
    plan = fixture.demotion_case(root)
    baseline = plan["baseline_commit"]
    result = {"scope": "actual_source_lifecycle_no_native_mocks", "mode": mode,

    if mode == "initial_moved":
        commit(root, mode)
    try:
        prepared = owner.prepare(root, root / "plan.json", root / "acceptance.json")
    except ValueError as error:
        assert mode == "initial_moved"
        result.update(prepare="refused", error=str(error), no_receipt=not (root / "receipt").exists())
    else:
        result["prepare"] = "passed"
        if mode.startswith("late_"):
            commit(root, mode, unrelated=mode == "late_unrelated")
        result["head_at_apply"] = fixture.run_git(root, "rev-parse", "HEAD")
        result["apply"] = owner.apply(prepared, root / "receipt", enabled=True)["state"]
        proof = owner.verify_completed(root, root / "plan.json", root / "receipt/application.json")
        result["initial_verify"] = proof["state"]
        if mode.startswith("historical_"):
            commit(root, mode, unrelated=mode.endswith("owners_and_unrelated"),
                   include_owners=mode.endswith("owners_and_unrelated"))
            if mode == "historical_tampered_current_owner":
                path = root / fixture.DEMOTED

                later = owner.verify_completed(root, root / "plan.json", root / "receipt/application.json")

                result.update(historical_verify="refused", historical_error=str(error))
            else:
                assert mode != "historical_tampered_current_owner"
                assert later == proof
                result.update(historical_verify=later["state"], historical_proof_exactly_unchanged=True)
        result["receipt_sha256"] = sha(root / "receipt/application.json")
    result.update(current_head=fixture.run_git(root, "rev-parse", "HEAD"),
