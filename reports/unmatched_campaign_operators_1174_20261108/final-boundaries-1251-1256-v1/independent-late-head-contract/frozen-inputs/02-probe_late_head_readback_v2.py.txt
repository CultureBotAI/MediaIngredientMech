import datetime
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import sys

R = Path.cwd()
P = R / '.temp/combined-16c-independent-v1'
F = P / 'frozen/repo'
O = P / 'root-probes'
os.environ.update(PYTHONDONTWRITEBYTECODE='1', GIT_OPTIONAL_LOCKS='0')
sys.path[:0] = [str(F / 'src'), str(F)]
spec = importlib.util.spec_from_file_location('exact_source_fixture_v2', F / 'tests/test_campaign_owner_apply.py')
fixture = importlib.util.module_from_spec(spec)
spec.loader.exec_module(fixture)
owner = fixture.owner

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

results = []
for mode in ['unchanged_positive', 'initial_moved_HEAD', 'late_empty_commit',
             'late_unrelated_tracked_commit', 'late_uncommitted_unrelated_tracked_change']:
    root = O / mode
    plan = json.loads((root / 'plan.json').read_text())
    baseline = plan['baseline_commit']
    result = {'mode': mode, 'fixture': str(root), 'accepted_baseline': baseline,
              'current_head': fixture.run_git(root, 'rev-parse', 'HEAD'),
              'plan_sha256': sha(root / 'plan.json'),
              'acceptance_sha256': sha(root / 'acceptance.json'),
              'source_module_sha256': sha(Path(owner.__file__)),
              'fixture_module_sha256': sha(Path(fixture.__file__)),
              'no_native_function_mocking': True}
    if mode == 'initial_moved_HEAD':
        try:
            owner.prepare(root, root / 'plan.json', root / 'acceptance.json')
        except ValueError as error:
            result.update(prepare='REFUSED', error=str(error), no_receipt=not (root / 'receipt').exists())
        else:
            raise AssertionError('initial stale HEAD unexpectedly accepted')
    elif mode == 'late_uncommitted_unrelated_tracked_change':
        prepared = owner.prepare(root, root / 'plan.json', root / 'acceptance.json')
        # The first harness stopped before touching this fixture because it
        # assumed a .gitignore existed. Use a newly staged unrelated file now.
        (root / 'unrelated-staged.txt').write_text('Not an accepted owner or output.\n')
        fixture.run_git(root, 'add', 'unrelated-staged.txt')
        try:
            owner.apply(prepared, root / 'receipt', enabled=True)
        except ValueError as error:
            result.update(prepare='PASS', apply='REFUSED', error=str(error))
        else:
            raise AssertionError('unrelated staged change unexpectedly accepted')
    else:
        # Earlier actual applications are retained, not repeated. Re-run the
        # real completed verifier over their unchanged receipts and artifacts.
        proof = owner.verify_completed(root, root / 'plan.json', root / 'receipt/application.json')
        result.update(prior_actual_apply_receipt='retained', verify_completed=proof['state'])
    receipt = root / 'receipt/application.json'
    if receipt.exists():
        saved = json.loads(receipt.read_text())
        result.update(receipt_state=saved['state'], receipt_baseline=saved['baseline_commit'],
                      recovery_required=saved['recovery_required'],
                      completed_operations=saved['completed_operations'],
                      receipt_sha256=sha(receipt))
    result.update(source_still_exists=(root / fixture.EDIT).exists(),
                  destination_exists=(root / fixture.DEMOTED).exists(),
                  literal_diff_from_baseline=fixture.run_git(root, 'diff', '--name-only', '-z', baseline, '--').split('\0'),
                  literal_diff_from_current_HEAD=fixture.run_git(root, 'diff', '--name-only', '-z', 'HEAD', '--').split('\0'))
    if mode == 'late_unrelated_tracked_commit':
        assert 'unrelated-after-acceptance.txt' in result['literal_diff_from_baseline']
        assert 'unrelated-after-acceptance.txt' not in result['literal_diff_from_current_HEAD']
        assert result['receipt_state'] == result['verify_completed'] == 'complete'
        assert result['current_head'] != result['accepted_baseline']
    results.append(result)

out = {'reviewed_head': '16c39b63bc95d3e7978ddccc6280b0d57954f4a5',
       'at': datetime.datetime.now(datetime.timezone.utc).isoformat(),
       'runtime': sys.version, 'python': sys.executable,
       'scope': 'Private actual demotion source lifecycle. Existing first four probe fixtures retained; completed readback repeated. Fifth probe uses staged unrelated file after real prepare. No native mocks, production source application, or scientific approval.',
       'first_harness_failure': {'script': 'provenance/probe_late_head.py',
                                'log': 'root-probe.log',
                                'cause': 'Fifth control assumed absent fixture .gitignore. First four actual results/receipts preserved, not reclassified as a clean first run.'},
       'results': results}
with (O / 'late-head-results-v2.json').open('x') as f:
    json.dump(out, f, indent=2)
    f.write('\n')
print(json.dumps(out, indent=2))
