from pathlib import Path
import datetime, hashlib, json, re

root = Path(__file__).resolve().parent
sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
receipts = {name: json.loads((root / (name + '-receipt.json')).read_bytes()) for name in ('qc', 'test')}
for receipt in receipts.values():
    assert receipt['exit_code'] == 0
    assert receipt['head'] == '16c39b63bc95d3e7978ddccc6280b0d57954f4a5'
    assert not receipt['source_differences'] and not receipt['tracked_status'] and not receipt['all_status']
    assert receipt['producer_unchanged'] and receipt['history_unchanged']
log = (root / 'test.log').read_text()
results = [line for line in log.splitlines() if re.search(r'\d+ passed.* in \d+', line)]
assert len(results) == 1, results
integration = [line for line in log.splitlines() if line.startswith('tests/test_campaign_demotion_integration.py ')]
assert len(integration) == 1 and '.........' in integration[0] and 's' not in integration[0].split('.py ', 1)[1]
coverage = [line for line in log.splitlines() if line.startswith('TOTAL') or 'fail-under' in line]
qc = (root / 'qc.log').read_text()
b4 = [line for line in qc.splitlines() if 'B4' in line or 'ontology' in line.lower() and 'skip' in line.lower()]
summary = {
    'at': datetime.datetime.now(datetime.UTC).isoformat(),
    'scope': 'Author full native validation at immutable historical16c checkpoint; no independent, CI, application or later-head approval',
    'head': receipts['test']['head'],
    'pytest_summary': results[0],
    'new_real_integration_cases': {'passed': 9, 'skipped': 0, 'actual_log_line': integration[0]},
    'coverage_log_lines': coverage,
    'rule_b4_limit': b4,
    'maintained_inputs_unchanged': len(json.loads((root / 'before.json').read_bytes())['source_sha256']),
    'producer_and_historical_inputs_unchanged': True,
    'clean_worktree': True,
    'receipts': {name: {'path': name + '-receipt.json', 'sha256': sha(root / (name + '-receipt.json')), 'log_sha256': sha(root / (name + '.log'))} for name in receipts},
    'limitations': ['Native default RuleB4 skipped where nested sibling ontology input path is absent; no gate or path topology changed.', 'Known1255/1256 corrections are separate private successors, not part of this16c validation.', 'Any later committed source head requires its own full validation.']
}
with (root / 'SUMMARY.json').open('x') as f: json.dump(summary, f, indent=2); f.write('\n')
names = ['before.json','runtime.json','run_checks.py','seal.py','qc-command.json','qc.log','qc-after.json','qc-receipt.json','test-command.json','test.log','test-after.json','test-receipt.json','SUMMARY.json']
manifest = {name: sha(root / name) for name in names}
with (root / 'MANIFEST.json').open('x') as f: json.dump({'schema_version': 1, 'files': manifest, 'private_fixture_trees_excluded': True}, f, indent=2); f.write('\n')
handoff = {'schema_version': 1, 'role': 'author_validation_only', 'head': summary['head'], 'manifest_sha256': sha(root / 'MANIFEST.json'), 'summary_sha256': sha(root / 'SUMMARY.json'), 'pytest_summary': summary['pytest_summary'], 'qc_exit': 0, 'full_just_test_exit': 0, 'new_real_integration_cases': summary['new_real_integration_cases'], 'clean_unchanged_inputs': True, 'future_head_requires_validation': True}
with (root / 'HANDOFF.json').open('x') as f: json.dump(handoff, f, indent=2); f.write('\n')
print(json.dumps({'handoff': sha(root / 'HANDOFF.json'), 'manifest': sha(root / 'MANIFEST.json'), 'summary': summary}, indent=2))
