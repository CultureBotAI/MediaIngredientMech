from pathlib import Path
import datetime
import hashlib
import json

import xml.etree.ElementTree as ET

D = Path(__file__).absolute().parent
C = D.parents[1]
def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
def read(path):
    return json.loads(path.read_text())
def save(name, value):
    with (D / name).open("x") as stream:
        json.dump(value, stream, indent=2, ensure_ascii=False)
        stream.write("\n")
def junit(name):
    suites = ET.parse(D / name).getroot().iter("testsuite")
    result = {key: 0 for key in ("tests", "errors", "failures", "skipped")}
    for suite in suites:
        for key in result:
            result[key] += int(suite.get(key, "0"))
    result["passed"] = result["tests"] - sum(result[key] for key in ("errors", "failures", "skipped"))
    return result

combined = junit("pytest-combined-v1.xml")
integration = junit("pytest-integration-v2.xml")

archive_checks = []
with tarfile.open(D / "baseline16c.tar") as archive:

        if not member.isfile():
            continue
        old = archive.extractfile(member).read()

        if member.name not in changed:
            assert current == old, member.name
        archive_checks.append({"path": member.name, "old_sha256": hashlib.sha256(old).hexdigest(),
                               "current_sha256": hashlib.sha256(current).hexdigest(), "changed": current != old})
assert {r["path"] for r in archive_checks if r["changed"]} == changed
save("archive-preservation-final.json", {"files": archive_checks, "only_two_proposed_files_differ": True})
